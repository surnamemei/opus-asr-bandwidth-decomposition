% Extra preamble for rendering the supplementary material (tools/render_ieee.sh with
% EXTRA_HEADER=tools/ieee-supplement.tex): a supplement is mostly tables and figures, so floats may
% also form float pages, several to a page (presentation only). Floats may also stand where they are
% defined (h), so tables stay beside their text; a float page must be at least 85 % full, so a
% part-filled page takes text below its floats instead of whitespace. The title still comes first
% (\suppressfloats[t] at the start of the supplement).
\makeatletter
\def\fps@table{htp}
\def\fps@figure{htp}
\makeatother
\setcounter{topnumber}{4}
\setcounter{dbltopnumber}{4}
\setcounter{totalnumber}{6}
\renewcommand{\topfraction}{0.95}
\renewcommand{\dbltopfraction}{0.95}
\renewcommand{\textfraction}{0.05}
\renewcommand{\floatpagefraction}{0.85}
\renewcommand{\dblfloatpagefraction}{0.85}
% Figures are set at 85 % of their natural width (presentation only): the supplement's figures are
% simple plots, and full-width figures took most of a page each in the one-column layout. The
% IEEE-FLOAT log line records the final width for tools/ieee_layout_report.py.
\makeatletter
\renewcommand\IEEEautofigure[3]{%
  \sbox\ieeebox{\includegraphics{#2}}%
  \setlength\ieeefinal{0.85\wd\ieeebox}%
  \typeout{IEEE-FLOAT kind=figure label=#1 placement=single-scaled natural=\the\wd\ieeebox\space
    height=\the\ht\ieeebox\space column=\the\columnwidth\space text=\the\textwidth\space final=\the\ieeefinal}%
  \begin{figure}\centering\includegraphics[width=\ieeefinal]{#2}\caption{#3}\label{#1}\end{figure}}
\makeatother
